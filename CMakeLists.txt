cmake_minimum_required(VERSION 3.24)
project(GrainSynthesizer VERSION 0.1.0 LANGUAGES CXX)
option(GRAIN_SYNTHESIZER_BUILD_TESTS "Build DSP and realtime bridge tests" ${PROJECT_IS_TOP_LEVEL})
add_library(grain_synthesizer_core STATIC src/grain_cloud.cpp src/RealtimeCloud.cpp)
add_library(GrainSynthesizer::core ALIAS grain_synthesizer_core)
target_include_directories(grain_synthesizer_core PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_compile_features(grain_synthesizer_core PUBLIC cxx_std_17)
set_target_properties(grain_synthesizer_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(GRAIN_SYNTHESIZER_BUILD_TESTS)
    enable_testing()
    add_test(NAME GrainEmitter.EventDrivenArchitecture
        COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/CheckEventDriven.cmake)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/googletest/CMakeLists.txt")
        add_subdirectory(external/googletest EXCLUDE_FROM_ALL)
    else()
        find_package(GTest CONFIG REQUIRED)
    endif()
    add_executable(grain_synthesizer_tests tests/upstream/grain_cloud_tests.cpp tests/RealtimeCloudTests.cpp)
    target_link_libraries(grain_synthesizer_tests PRIVATE GrainSynthesizer::core GTest::gtest_main)
    target_compile_features(grain_synthesizer_tests PRIVATE cxx_std_20)
    include(GoogleTest)
    gtest_discover_tests(grain_synthesizer_tests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/miniaudio/CMakeLists.txt")
        enable_language(C)
        set(MINIAUDIO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(MINIAUDIO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        add_subdirectory(external/miniaudio EXCLUDE_FROM_ALL)
        add_executable(grain_miniaudio_tests tests/MiniAudioBridgeTests.cpp)
        target_include_directories(grain_miniaudio_tests PRIVATE integrations/miniaudio)
        target_link_libraries(grain_miniaudio_tests PRIVATE GrainSynthesizer::core miniaudio GTest::gtest_main)
        gtest_discover_tests(grain_miniaudio_tests)
    endif()
endif()
